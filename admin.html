TRACDI MSA — Msimamizi
Tazama Tovuti
Supabase haijawekwa. Fungua admin.html, weka url na anonKey (publishable/anon key pekee, kamwe usiweke service-role key) kwenye CONFIG. Tumia thamani zilezile ulizoweka kwenye index.html.

Ingia kama Msimamizi

Akaunti hii haina ruhusa ya msimamizi. Mmiliki lazima aiongeze kwenye jedwali la admins (angalia maelekezo ya kuanzisha hapa chini).

Maelekezo ya kuanzisha Supabase (mara moja tu)
  1. Fungua Supabase → SQL Editor, bandika SQL ifuatayo na ubonyeze Run. Inatengeneza majedwali, sera za RLS na hifadhi ya picha (media).
  2. Supabase → Authentication → Users → Add user: weka barua pepe na nenosiri la msimamizi (washa “Auto confirm”).
  3. Kisha endesha: insert into public.admins (user_id) select id from auth.users where email = 'BARUA@YAKO.COM';
  4. Kwa usalama, zima usajili wa watu wapya: Authentication → Sign In / Providers → Email → zima “Allow new users to sign up”. 
-- ========== 1. WASIMAMIZI ==========
create table if not exists public.admins (
  user_id uuid primary key references auth.users(id) on delete cascade,
  created_at timestamptz not null default now()
);
alter table public.admins enable row level security;

create or replace function public.is_admin()
returns boolean language sql stable security definer set search_path = public as $$
  select exists (select 1 from public.admins where user_id = auth.uid());
$$;
revoke all on function public.is_admin() from public;
grant execute on function public.is_admin() to authenticated;

drop policy if exists "admin anajiona" on public.admins;
create policy "admin anajiona" on public.admins for select to authenticated using (user_id = auth.uid());
grant select on public.admins to authenticated;

-- ========== 2. MAJEDWALI YA MAUDHUI ==========
create table if not exists public.announcements (
  id uuid primary key default gen_random_uuid(), created_at timestamptz not null default now(),
  published boolean not null default true,
  title text not null, category text not null default 'Jumla', post_date date not null default current_date,
  description text not null, image_url text, link_url text, link_label text);

create table if not exists public.events (
  id uuid primary key default gen_random_uuid(), created_at timestamptz not null default now(),
  published boolean not null default true,
  title text not null, category text not null, event_date date not null, event_time time,
  location text, description text, image_url text);

create table if not exists public.gallery (
  id uuid primary key default gen_random_uuid(), created_at timestamptz not null default now(),
  published boolean not null default true,
  image_url text not null, alt text not null, caption text, description text);

create table if not exists public.quran (
  id uuid primary key default gen_random_uuid(), created_at timestamptz not null default now(),
  published boolean not null default true,
  surah text not null, surah_number int not null, ayah text not null, arabic text not null, swahili text not null);

create table if not exists public.hadith (
  id uuid primary key default gen_random_uuid(), created_at timestamptz not null default now(),
  published boolean not null default true,
  topic text not null, arabic text, swahili text not null, source text not null, grade text);

create table if not exists public.duas (
  id uuid primary key default gen_random_uuid(), created_at timestamptz not null default now(),
  published boolean not null default true,
  category text not null, title text not null, arabic text not null, translit text, swahili text not null, source text);

create table if not exists public.reminders (
  id uuid primary key default gen_random_uuid(), created_at timestamptz not null default now(),
  published boolean not null default true,
  topic text not null, body text not null, ref text);

create table if not exists public.qa (
  id uuid primary key default gen_random_uuid(), created_at timestamptz not null default now(),
  published boolean not null default true,
  question text not null, answer text not null, reviewer text, source text);

create table if not exists public.leadership (
  id uuid primary key default gen_random_uuid(), created_at timestamptz not null default now(),
  published boolean not null default true,
  name text not null, role text not null, photo_url text, sort_order int not null default 100);

create table if not exists public.calendar_dates (
  id uuid primary key default gen_random_uuid(), created_at timestamptz not null default now(),
  published boolean not null default true,
  title text not null, event_date date not null, note text);

-- ========== 3. RLS: umma husoma yaliyochapishwa tu; msimamizi anafanya yote ==========
do $$
declare t text;
begin
  foreach t in array array['announcements','events','gallery','quran','hadith','duas','reminders','qa','leadership','calendar_dates'] loop
    execute format('alter table public.%I enable row level security', t);
    execute format('drop policy if exists "umma anasoma yaliyochapishwa" on public.%I', t);
    execute format('create policy "umma anasoma yaliyochapishwa" on public.%I for select to anon, authenticated using (published = true)', t);
    execute format('drop policy if exists "msimamizi ana ruhusa kamili" on public.%I', t);
    execute format('create policy "msimamizi ana ruhusa kamili" on public.%I for all to authenticated using (public.is_admin()) with check (public.is_admin())', t);
    execute format('revoke all on public.%I from anon, authenticated', t);
    execute format('grant select on public.%I to anon', t);
    execute format('grant select, insert, update, delete on public.%I to authenticated', t);
  end loop;
end $$;

-- ========== 4. UJUMBE WA WAGENI (submissions) ==========
create table if not exists public.submissions (
  id uuid primary key default gen_random_uuid(), created_at timestamptz not null default now(),
  name text check (name is null or char_length(name) <= 80),
  topic text not null check (char_length(topic) between 1 and 60),
  message text not null check (char_length(message) between 10 and 1000),
  contact text check (contact is null or char_length(contact) <= 120));
alter table public.submissions add column if not exists handled boolean not null default false;
alter table public.submissions enable row level security;
grant insert on public.submissions to anon;
grant select, update, delete on public.submissions to authenticated;

drop policy if exists "umma anaweza kutuma tu" on public.submissions;
create policy "umma anaweza kutuma tu" on public.submissions for insert to anon
  with check (char_length(message) between 10 and 1000);
drop policy if exists "msimamizi anasoma ujumbe" on public.submissions;
create policy "msimamizi anasoma ujumbe" on public.submissions for select to authenticated using (public.is_admin());
drop policy if exists "msimamizi anasasisha ujumbe" on public.submissions;
create policy "msimamizi anasasisha ujumbe" on public.submissions for update to authenticated using (public.is_admin()) with check (public.is_admin());
drop policy if exists "msimamizi anafuta ujumbe" on public.submissions;
create policy "msimamizi anafuta ujumbe" on public.submissions for delete to authenticated using (public.is_admin());

-- ========== 5. HIFADHI YA PICHA (bucket "media") ==========
insert into storage.buckets (id, name, public, file_size_limit, allowed_mime_types)
values ('media','media', true, 5242880, array['image/jpeg','image/png','image/webp','image/gif'])
on conflict (id) do update set public = true, file_size_limit = 5242880,
  allowed_mime_types = array['image/jpeg','image/png','image/webp','image/gif'];

drop policy if exists "msimamizi anapakia picha" on storage.objects;
create policy "msimamizi anapakia picha" on storage.objects for insert to authenticated
  with check (bucket_id = 'media' and public.is_admin());
drop policy if exists "msimamizi anasasisha picha" on storage.objects;
create policy "msimamizi anasasisha picha" on storage.objects for update to authenticated
  using (bucket_id = 'media' and public.is_admin()) with check (bucket_id = 'media' and public.is_admin());
drop policy if exists "msimamizi anafuta picha" on storage.objects;
create policy "msimamizi anafuta picha" on storage.objects for delete to authenticated
  using (bucket_id = 'media' and public.is_admin());
